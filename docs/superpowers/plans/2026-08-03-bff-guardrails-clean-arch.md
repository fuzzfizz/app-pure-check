# Implementation Plan: BFF Pattern, Guardrails & Clean Architecture Refactoring

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement BFF Pattern for Gemini API key security, Database Guardrails (Blacklist) for hazardous chemicals, Clean Architecture (Repository Pattern) for scan functionality, and Health Disclaimers on the result screen.

**Architecture:** Refactor `scan_provider.dart` to use an abstract `ScanRepository` implemented by `ScanRepositoryImpl`. Route ingredient analysis requests through Supabase Edge Function `analyze-ingredients` (which cross-checks against a `hazardous_chemicals` blacklist table and calls Gemini API securely on the server), falling back to client-side `GeminiService` if offline/local dev. Render a health disclaimer banner on `result_screen.dart`.

**Tech Stack:** Flutter, Riverpod, Supabase (Database, Auth, Edge Functions), Dart, TypeScript (Deno for Edge Functions).

## Global Constraints

- Flutter framework version compatibility (Dart 3.x+).
- Clean Architecture separation: UI components (`screens`) communicate only with StateNotifiers (`providers`), which communicate only with Domain Repositories (`repositories`), which interact with Data Sources (`services` / Edge Functions).
- Backward compatibility: Preserves `GeminiService` for local key testing/fallback without breaking existing setups.

```dart
import 'package:flutter_riverpod/flutter_riverpod.dart';
import '../../../../core/models/product.dart';
import '../../../../core/models/analysis_result.dart';
import '../../../../core/models/user_profile.dart';
import '../../../../core/models/allergen.dart';
import '../../../../core/services/supabase_service.dart';
import '../../../../core/services/beauty_facts_service.dart';
import '../../../../core/services/gemini_service.dart';
import '../../domain/repositories/scan_repository.dart';
import '../../../auth/providers/auth_provider.dart';

class ScanRepositoryImpl implements ScanRepository {
  final SupabaseService supabaseService;
  final BeautyFactsService beautyFactsService;
  final GeminiService geminiService;

  ScanRepositoryImpl({
    required this.supabaseService,
    required this.beautyFactsService,
    required this.geminiService,
  });

  @override
  Future<Product?> fetchProductByBarcode(String barcode) async {
    var product = await supabaseService.getProductByBarcode(barcode);
    product ??= await beautyFactsService.fetchByBarcode(barcode);
    return product;
  }

  @override
  Future<Product> upsertProduct(Product product) async {
    return await supabaseService.upsertProduct(product);
  }

  @override
  Future<AnalysisResult> analyzeIngredients({
    required UserProfile profile,
    required List<Allergen> allergens,
    required List<String> ingredients,
  }) async {
    try {
      // 1. Try BFF (Supabase Edge Function)
      final client = supabaseService.client;
      final response = await client.functions.invoke(
        'analyze-ingredients',
        body: {
          'profile': profile.toJson(),
          'allergens': allergens.map((a) => a.ingredientName).toList(),
          'ingredients': ingredients,
        },
      );

      if (response.status == 200 && response.data != null) {
        final data = response.data as Map<String, dynamic>;
        return AnalysisResult.fromJson(data);
      }
    } catch (_) {
      // Edge Function error or offline fallback to client GeminiService
    }

    // 2. Fallback to direct GeminiService
    return await geminiService.analyzeIngredients(
      profile: profile,
      allergens: allergens,
      ingredients: ingredients,
    );
  }

  @override
  Future<void> saveScanHistory({
    required String userId,
    required String productId,
    required AnalysisResult result,
  }) async {
    await supabaseService.addScanHistory(
      userId: userId,
      productId: productId,
      result: result,
    );
  }
}

final scanRepositoryProvider = Provider<ScanRepository>((ref) {
  return ScanRepositoryImpl(
    supabaseService: ref.watch(supabaseServiceProvider),
    beautyFactsService: BeautyFactsService(),
    geminiService: GeminiService(),
  );
});
```

```dart
import 'package:flutter_riverpod/flutter_riverpod.dart';
import '../../../core/models/product.dart';
import '../../../core/models/analysis_result.dart';
import '../../auth/providers/auth_provider.dart';
import '../domain/repositories/scan_repository.dart';
import '../data/repositories/scan_repository_impl.dart';

enum ScanStep { idle, scanning, fetching, verifying, manualEntry, analyzing, error }

class ScanState {
  final ScanStep step;
  final String? barcode;
  final Product? product;
  final AnalysisResult? analysisResult;
  final String? error;

  ScanState({
    this.step = ScanStep.idle,
    this.barcode,
    this.product,
    this.analysisResult,
    this.error,
  });

  ScanState copyWith({
    ScanStep? step,
    String? barcode,
    Product? product,
    AnalysisResult? analysisResult,
    String? error,
  }) {
    return ScanState(
      step: step ?? this.step,
      barcode: barcode ?? this.barcode,
      product: product ?? this.product,
      analysisResult: analysisResult ?? this.analysisResult,
      error: error,
    );
  }
}

class ScanNotifier extends StateNotifier<ScanState> {
  final Ref ref;

  ScanNotifier(this.ref) : super(ScanState());

  ScanRepository get _repository => ref.read(scanRepositoryProvider);

  void reset() {
    state = ScanState();
  }

  void startScanning() {
    state = ScanState(step: ScanStep.scanning);
  }

  Future<void> onBarcodeScanned(String barcode) async {
    state = state.copyWith(step: ScanStep.fetching, barcode: barcode);
    try {
      final product = await _repository.fetchProductByBarcode(barcode);

      if (product != null) {
        state = state.copyWith(step: ScanStep.verifying, product: product);
      } else {
        state = state.copyWith(
          step: ScanStep.manualEntry,
          product: Product(
            id: '',
            barcode: barcode,
            name: '',
            ingredients: [],
            source: ProductSource.userEntered,
          ),
        );
      }
    } catch (e) {
      state = state.copyWith(step: ScanStep.error, error: e.toString());
    }
  }

  void setManualProduct(Product product) {
    state = state.copyWith(step: ScanStep.verifying, product: product);
  }

  Future<AnalysisResult?> analyzeAndSave(Product finalProduct) async {
    state = state.copyWith(step: ScanStep.analyzing, product: finalProduct);
    try {
      final user = ref.read(currentUserProvider);
      if (user == null) throw Exception('User not logged in');

      final supabaseService = ref.read(supabaseServiceProvider);

      // Save/update product via Repository
      final savedProduct = await _repository.upsertProduct(finalProduct);

      // Fetch user profile and allergens
      final profile = await supabaseService.getProfile(user.id);
      final allergens = await supabaseService.getAllergens(user.id);

      if (profile == null) throw Exception('Profile not found');

      // Analyze ingredients via Repository (BFF / Edge Function)
      final analysis = await _repository.analyzeIngredients(
        profile: profile,
        allergens: allergens,
        ingredients: savedProduct.ingredients,
      );

      // Save scan history via Repository
      await _repository.saveScanHistory(
        userId: user.id,
        productId: savedProduct.id,
        result: analysis,
      );

      state = state.copyWith(step: ScanStep.idle, analysisResult: analysis, product: savedProduct);
      return analysis;
    } catch (e) {
      state = state.copyWith(step: ScanStep.error, error: e.toString());
      return null;
    }
  }
}

final scanNotifierProvider = StateNotifierProvider<ScanNotifier, ScanState>((ref) {
  return ScanNotifier(ref);
});
```

Create `supabase/functions/analyze-ingredients/index.ts`:
```typescript
import { serve } from "https://deno.land/std@0.168.0/http/server.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2.39.8";

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  try {
    const { profile, allergens, ingredients } = await req.json();

    const supabaseUrl = Deno.env.get('SUPABASE_URL') ?? '';
    const supabaseServiceKey = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY') ?? '';
    const geminiApiKey = Deno.env.get('GEMINI_API_KEY') ?? '';

    const supabase = createClient(supabaseUrl, supabaseServiceKey);

    // 1. Guardrails Check: Query hazardous_chemicals table
    const { data: hazardousList } = await supabase
      .from('hazardous_chemicals')
      .select('chemical_name, aliases, risk_level, hazard_category, description_th');

    const flaggedFromBlacklist: Array<{ name: string; reason: string; risk_level: string }> = [];

    if (hazardousList && Array.isArray(ingredients)) {
      for (const ing of ingredients) {
        const lowerIng = String(ing).toLowerCase();
        for (const item of hazardousList) {
          const matchName = item.chemical_name.toLowerCase();
          const matchAlias = (item.aliases || []).some((a: string) => a.toLowerCase() === lowerIng);
          if (lowerIng.includes(matchName) || matchAlias) {
            flaggedFromBlacklist.push({
              name: ing,
              reason: item.description_th || `พบในรายการสารอันตราย (${item.hazard_category})`,
              risk_level: item.risk_level || 'danger',
            });
          }
        }
      }
    }

    // 2. Call Gemini API securely with server-side key
    let aiResponse = {
      overall_safety: flaggedFromBlacklist.length > 0 ? "danger" : "safe",
      summary_th: "วิเคราะห์ส่วนผสมเรียบร้อยแล้ว",
      summary_en: "Ingredients analysis complete",
      flagged_ingredients: flaggedFromBlacklist,
      ingredient_breakdown: (ingredients || []).map((i: string) => ({
        name: i,
        function: "ส่วนผสมทางเครื่องสำอาง",
        risk_level: flaggedFromBlacklist.some(f => f.name === i) ? "danger" : "safe"
      }))
    };

    if (geminiApiKey) {
      const prompt = `
Analyze cosmetic ingredients: ${JSON.stringify(ingredients)}
User profile: ${JSON.stringify(profile)}
Allergens: ${JSON.stringify(allergens)}
Blacklisted found: ${JSON.stringify(flaggedFromBlacklist)}

Return strictly JSON:
{
  "overall_safety": "safe|caution|danger",
  "summary_th": "คำอธิบายภาษาไทย",
  "summary_en": "English summary",
  "flagged_ingredients": [{"name": "", "reason": "", "risk_level": "caution|danger"}],
  "ingredient_breakdown": [{"name": "", "function": "", "risk_level": "safe|caution|danger"}]
}
`;
      const geminiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${geminiApiKey}`;
      const res = await fetch(geminiUrl, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          contents: [{ parts: [{ text: prompt }] }],
          generationConfig: { responseMimeType: "application/json" }
        })
      });

      if (res.ok) {
        const jsonRes = await res.json();
        const rawText = jsonRes.candidates?.[0]?.content?.parts?.[0]?.text;
        if (rawText) {
          try {
            aiResponse = JSON.parse(rawText);
            // Merge blacklist flags if AI missed them
            if (flaggedFromBlacklist.length > 0) {
              aiResponse.overall_safety = "danger";
              for (const flag of flaggedFromBlacklist) {
                if (!aiResponse.flagged_ingredients.some((f: any) => f.name === flag.name)) {
                  aiResponse.flagged_ingredients.push(flag);
                }
              }
            }
          } catch (_) {}
        }
      }
    }

    return new Response(JSON.stringify(aiResponse), {
      headers: { ...corsHeaders, 'Content-Type': 'application/json' },
      status: 200,
    });
  } catch (err: any) {
    return new Response(JSON.stringify({ error: err.message }), {

});
```
